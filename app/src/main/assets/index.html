  1. 打开 supabase.com 免费注册，新建一个 project。
  2. 左侧菜单 Settings → API，复制 Project URL 和 anon public key，分别填到上面两个框。
  3. 左侧菜单 SQL Editor，粘贴下面这段执行一次（建表 + 开放读写权限）：
  4. 点「保存并检测」，提示连通正常后，回到上面点「开启」或「加入」即可。
create table if not exists wj_docs (
  id text primary key,
  token text not null,
  v int not null default 1,
  ts bigint not null default 0,
  data jsonb not null
);
alter table wj_docs enable row level security;
create policy "allow all" on wj_docs for all using (true) with check (true);
